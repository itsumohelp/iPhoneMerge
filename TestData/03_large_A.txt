015:
016:     func add(_ user: User) {
017:         users.append(user)
018:     }
019:
020:     func find(by id: Int) -> User? {
021:         return users.first { $0.id == id }
022:     }
023:
024:     func all() -> [User] {
025:         return users
026:     }
027:
